/**
 * Researcher view ("fog of data", PLAN §5.10): what an analyst with Inside Airbnb-style snapshots
 * and a CPI series would see. Pure post-processing of a run's true series, so it never changes
 * model dynamics. Draws are keyed hashes (seed, zone, tick, purpose): deterministic and CRN-safe.
 *
 * Listings: at each snapshot tick, observed_z = true_z × completeness_t × exp(σ_z·ε − σ_z²/2),
 * where completeness_t ~ N(mean, sd) is common to all zones (scrape coverage). With probability
 * truncationProb a snapshot is truncated: each zone keeps only U(keepMin, keepMax) of its
 * listings. A snapshot is flagged incomplete when its total is below flagThreshold × the mean of
 * its neighbouring snapshots' totals (the first project's rule), and analysts drop it.
 *
 * CPI: a quarterly reading of the CPI-style index plus N(0, cpiNoiseSd) points.
 */
import observation from './params/observation.json' with { type: 'json' };
import { normal, Purpose, uniform } from './rng.ts';
import type { RunOutput } from './types.ts';

export interface ObservationParams {
  completenessMean: number;
  completenessSd: number;
  zoneNoiseSd: number;
  truncationProb: number;
  truncationKeepMin: number;
  truncationKeepMax: number;
  flagThreshold: number;
  cpiNoiseSd: number;
}

export const DEFAULT_OBSERVATION: Readonly<ObservationParams> = observation as ObservationParams;

export interface ResearcherPanel {
  snapshotTicks: number[];
  /** Observed active listings in real units, [zone][snapshot]. */
  observed: number[][];
  /** True listings in real units, [zone][snapshot]. */
  truth: number[][];
  /** Snapshot flagged as incomplete (truncated). */
  incomplete: boolean[];
}

export function researcherPanel(
  out: RunOutput,
  seed: number,
  snapshotTicks: readonly number[],
  obs: ObservationParams = DEFAULT_OBSERVATION,
): ResearcherPanel {
  const f = out.meta.firstTick;
  const scale = out.meta.scale;
  const zones = out.series.zones;
  const truth = zones.map((z) => snapshotTicks.map((t) => z.strListings[t - f]! / scale));
  const observed = zones.map(() => snapshotTicks.map(() => 0));
  snapshotTicks.forEach((t, k) => {
    const completeness = Math.min(
      1,
      Math.max(0.5, obs.completenessMean + obs.completenessSd * normal(seed, 0, t, Purpose.OBS_COMPLETE)),
    );
    const truncated = uniform(seed, 0, t, Purpose.OBS_TRUNCATED) < obs.truncationProb;
    zones.forEach((_, zi) => {
      const s = obs.zoneNoiseSd;
      let v = truth[zi]![k]! * completeness * Math.exp(s * normal(seed, zi + 1, t, Purpose.OBS_ZONE) - (s * s) / 2);
      if (truncated) {
        const keep = obs.truncationKeepMin + (obs.truncationKeepMax - obs.truncationKeepMin) * uniform(seed, zi + 1, t, Purpose.OBS_TRUNCATED);
        v *= keep;
      }
      observed[zi]![k] = Math.round(v);
    });
  });
  const totals = snapshotTicks.map((_, k) => observed.reduce((a, z) => a + z[k]!, 0));
  const incomplete = totals.map((tot, k) => {
    const nb = [totals[k - 1], totals[k + 1]].filter((x): x is number => x !== undefined);
    const ref = nb.reduce((a, b) => a + b, 0) / Math.max(1, nb.length);
    return nb.length > 0 && tot < obs.flagThreshold * ref;
  });
  return { snapshotTicks: [...snapshotTicks], observed, truth, incomplete };
}

/** Quarterly noisy CPI-style readings (every 3rd tick from tick 0), as published series would be. */
export function researcherCpi(
  out: RunOutput,
  seed: number,
  obs: ObservationParams = DEFAULT_OBSERVATION,
): { tick: number[]; value: number[] } {
  const tick: number[] = [];
  const value: number[] = [];
  out.series.tick.forEach((t, i) => {
    if (t < 0 || t % 3 !== 0) return;
    tick.push(t);
    value.push(out.series.cpiRentIndex[i]! + obs.cpiNoiseSd * normal(seed, 0, t, Purpose.OBS_CPI));
  });
  return { tick, value };
}
